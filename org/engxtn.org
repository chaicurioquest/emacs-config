* Ensure TeX Binaries Available
#+BEGIN_SRC emacs-lisp
(dolist (dir '("/usr/local/texlive/2025/bin/x86_64-linux"
               "/usr/bin"
               "/usr/local/bin"))
  (when (file-directory-p dir)
    (add-to-list 'exec-path dir)))

;; Only add the TeX Live path that is likely missing from the shell PATH
(let ((texlive-bin "/usr/local/texlive/2025/bin/x86_64-linux"))
  (when (and (file-directory-p texlive-bin)
             (not (string-match-p (regexp-quote texlive-bin)
                                  (or (getenv "PATH") ""))))
    (setenv "PATH" (concat texlive-bin ":" (getenv "PATH")))))

* Theme-Aware Preview Refresh
#+BEGIN_SRC emacs-lisp
(defun my/org-latex-preview-refresh ()
  "Clear preview cache and refresh when theme changes."
  (interactive)
    (when (file-directory-p org-preview-latex-image-directory)
      (delete-directory org-preview-latex-image-directory t))
        (when (derived-mode-p 'org-mode)
          (org-latex-preview '(64))))

(add-hook 'after-load-theme-hook #'my/org-latex-preview-refresh)
#+END_SRC

* Tikz/CircuitTikZ Integrations
#+BEGIN_SRC emacs-lisp
;; ── TikZ/CircuiTikZ → PNG pipeline ───────────────────────────────────────
;; Call from any Org emacs-lisp block:
;;   (my/tikz-to-png "fig-name" "\\begin{circuitikz}...\\end{circuitikz}")
(defun my/tikz-to-png (name tikz-body)
  "Compile TIKZ-BODY to images/NAME.png beside the current org file.
Scratch files (.tex .pdf .log) go to /tmp/org-babel/.
Returns relative path from the org file directory."
  (let* ((tmp-dir  "/tmp/org-babel/")
         (out-dir  (expand-file-name
                    "images/"
                    (file-name-directory
                     (or (buffer-file-name) default-directory))))
         (tex-file (concat tmp-dir name ".tex"))
         (pdf-file (concat tmp-dir name ".pdf"))
         (log-file (concat tmp-dir name ".log"))
         (tmp-png  (concat tmp-dir name ".png"))   ; ← -singlefile output
         (out-png  (expand-file-name (concat name ".png") out-dir)))
    (unless (file-directory-p tmp-dir) (make-directory tmp-dir t))
    (unless (file-directory-p out-dir) (make-directory out-dir t))
    (dolist (f (list pdf-file out-png log-file))
      (when (file-exists-p f) (delete-file f)))
    (with-temp-file tex-file
      (insert "\\documentclass[border=5mm]{standalone}\n"
              "\\usepackage{tikz}\n"
              "\\usetikzlibrary{calc,positioning}\n"
              "\\usepackage[american,emptypmoscircle]{circuitikz}\n"
              "\\usepackage{pgfplots}\n"
              "\\usepgfplotslibrary{fillbetween}\n" 
              "\\pgfplotsset{compat=1.18, lua backend=false}\n"
              "\\begin{document}\n"
              tikz-body "\n"
              "\\end{document}\n"))
    (let ((rc (call-process "lualatex" nil `(:file ,log-file) nil
                            "-interaction=nonstopmode"
                            (concat "-output-directory=" tmp-dir)
                            tex-file)))
      (unless (zerop rc)
        (error "lualatex failed (rc=%d). See %s" rc log-file)))
    (let ((rc (call-process "pdftoppm" nil nil nil   ; ← -singlefile: no wildcard needed
                            "-png" "-r" "300" "-singlefile"
                            pdf-file (concat tmp-dir name))))
      (unless (zerop rc)
        (error "pdftoppm failed (rc=%d)" rc)))
    (rename-file tmp-png out-png t)
    (file-relative-name out-png
                        (file-name-directory
                         (or (buffer-file-name) default-directory)))))
#+END_SRC
