  (when (and (file-directory-p texlive-bin)
             (not (string-match-p (regexp-quote texlive-bin)

* Theme-Aware Preview Refresh
#+BEGIN_SRC emacs-lisp
(defun my/org-latex-preview-refresh ()
  "Clear preview cache and refresh when theme changes."
  (interactive)
    (when (file-directory-p org-preview-latex-image-directory)
      (delete-directory org-preview-latex-image-directory t))
        (when (derived-mode-p 'org-mode)
          (org-latex-preview '(64))))

(add-hook 'after-load-theme-hook #'my/org-latex-preview-refresh)
#+END_SRC

* Tikz/CircuitTikZ Integrations
#+BEGIN_SRC emacs-lisp
;; ── TikZ/CircuiTikZ → PNG pipeline ───────────────────────────────────────
;; Call from any Org emacs-lisp block:
;;   (my/tikz-to-png "name" body)
;;     → PNG to images/NAME.png beside .org  [inline display + LaTeX export]
;;   (my/tikz-to-png "name" body nil "/wspace/.../name.pdf")
;;     → PNG to images/NAME.png beside .org  AND  PDF to absolute path
;;   (my/tikz-to-png "name" body "/abs/path/name.png")
;;     → PNG to absolute path  (returns absolute path to #+RESULTS:)
;;   (my/tikz-to-png "name" body "/abs/path/name.png" "/abs/path/name.pdf")
;;     → PNG + PDF both to absolute paths
;;
;; Scratch files (.tex .pdf .log) always go to /tmp/org-babel/.
;; All existing 2-arg callers (inverter2, etc.) are unaffected.
(defun my/tikz-to-png (name tikz-body &optional png-out-path pdf-out-path)
  "Compile TIKZ-BODY → PNG, optionally also copying PDF to an absolute path.
NAME         : base filename, no extension (e.g. \"inverter2\").
TIKZ-BODY    : raw LaTeX string starting with \\begin{...}.
PNG-OUT-PATH : absolute PNG destination.  nil = images/NAME.png beside .org.
PDF-OUT-PATH : when non-nil, copy intermediate PDF to this absolute path.
Returns relative PNG path when beside .org, absolute path otherwise."
  (let* ((tmp-dir  "/tmp/org-babel/")
         (org-dir  (file-name-directory
                    (or (buffer-file-name) default-directory)))
         (out-dir  (expand-file-name "images/" org-dir))
         (tex-file (concat tmp-dir name ".tex"))
         (pdf-file (concat tmp-dir name ".pdf"))
         (log-file (concat tmp-dir name ".log"))
         (tmp-png  (concat tmp-dir name ".png"))   ; pdftoppm -singlefile output
         (out-png  (or png-out-path
                       (expand-file-name (concat name ".png") out-dir))))
    ;; Ensure directories exist
    (unless (file-directory-p tmp-dir)
      (make-directory tmp-dir t))
    (unless (file-directory-p (file-name-directory out-png))
      (make-directory (file-name-directory out-png) t))
    ;; Clean stale outputs
    (dolist (f (list pdf-file out-png log-file))
      (when (file-exists-p f) (delete-file f)))
    ;; Write standalone .tex
    (with-temp-file tex-file
      (insert "\\documentclass[border=5mm]{standalone}\n"
              "\\usepackage{tikz}\n"
              "\\usetikzlibrary{calc,positioning,automata,arrows.meta,fit}\n"
              "\\usepackage[american,emptypmoscircle]{circuitikz}\n"
              "\\usepackage{pgfplots}\n"
              "\\usepgfplotslibrary{fillbetween}\n"
              "\\pgfplotsset{compat=1.18, lua backend=false}\n"
              "\\begin{document}\n"
              tikz-body "\n"
              "\\end{document}\n"))
    ;; Compile → PDF
    (let ((rc (call-process "lualatex" nil `(:file ,log-file) nil
                            "-interaction=nonstopmode"
                            (concat "-output-directory=" tmp-dir)
                            tex-file)))
      (unless (zerop rc)
        (error "my/tikz-to-png: lualatex failed (rc=%d). See %s" rc log-file)))
    ;; Convert PDF → PNG at 300 dpi
    (let ((rc (call-process "pdftoppm" nil nil nil
                            "-png" "-r" "300" "-singlefile"
                            pdf-file (concat tmp-dir name))))
      (unless (zerop rc)
        (error "my/tikz-to-png: pdftoppm failed (rc=%d)" rc)))
    ;; Move PNG to final destination
    (rename-file tmp-png out-png t)
    ;; Optionally copy PDF sidecar to absolute path (external docs tooling)
    (when (and pdf-out-path (file-exists-p pdf-file))
      (unless (file-directory-p (file-name-directory pdf-out-path))
        (make-directory (file-name-directory pdf-out-path) t))
      (copy-file pdf-file pdf-out-path t))
    ;; Return relative path when PNG is beside .org, else absolute
    (let ((rel (if png-out-path
               out-png
             (file-relative-name out-png org-dir))))
                (format "[[file:%s]]" rel))))
