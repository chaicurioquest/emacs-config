;; Only configure if on laptop, mu is available, and my-maildir is defined
(when (and (eq my-device 'laptop)
           (executable-find "mu")
           (boundp 'my-maildir))

;; Set core vars EARLY — before mu4e load/validation/timers
  (setq mu4e-maildir                my-maildir
        mu4e-create-maildir-if-needed nil
        mu4e-get-mail-command       "true"
        mu4e-update-interval        0
        mu4e-headers-auto-update    nil
        mu4e-view-prefer-html       t
        mu4e-headers-include-related t
        mu4e-view-show-images       t
        mu4e-inbox-folder           "/INBOX")

  (message "mail.el: mu4e-maildir set to %s" mu4e-maildir)

    :straight nil
    :config
    (setq mu4e-sent-folder   "/[Gmail]/Sent Mail"
          mu4e-drafts-folder "/[Gmail]/Drafts"
          mu4e-trash-folder  "/[Gmail]/Bin"
          mu4e-maildir-shortcuts
          '(("/INBOX"              . ?i)
            ("/[Gmail]/Sent Mail"  . ?s)
            ("/[Gmail]/Bin"        . ?t)
            ("/[Gmail]/Drafts"     . ?d)))

    ;; Cancel refresh timer — using mbsync externally
    (when (boundp 'mu4e~refresh-timer)
      (cancel-timer mu4e~refresh-timer)
      (setq mu4e~refresh-timer nil)
      (message "mail.el: mu4e refresh timer disabled (using mbsync)")))

;; mu4e-org integration
  (use-package mu4e-org
    :defer t
    :after mu4e)

  (with-eval-after-load 'mu4e-view
    (define-key mu4e-view-mode-map (kbd "C-c C-c") #'mu4e-org-store-and-capture))

;; Org capture template for mu4e messages
  (with-eval-after-load 'org
    (require 'org-capture)
    (setq org-capture-templates
          (append (bound-and-true-p org-capture-templates)
                  `(("m" "Email task (mu4e)" entry
                     (file+headline ,(expand-file-name "inbox.org" my/notes-root-dir) "Emails")
                     "* TODO [#A] %:subject :email:\n:PROPERTIES:\n:From: %:from\n:Message-ID: %:message-id\n:END:\nLink: [[mu4e:msgid:%:message-id][Open email]]\n%?\n"
                     :empty-lines 1 :prepend t)))))
#+END_SRC

#+BEGIN_SRC emacs-lisp
;; mu4e-alert — notifications for unread INBOX (laptop only)
  (use-package mu4e-alert
    :straight (:host github :repo "iqbalansari/mu4e-alert")
    :defer t
    :after mu4e
    :config
    (mu4e-alert-enable-mode-line-display)
    (mu4e-alert-set-default-style 'libnotify)
    (setq mu4e-alert-interesting-mail-query "maildir:/INBOX AND flag:unread"))

  (with-eval-after-load 'org
    (defun my/mu4e-alert-enable ()
      "Enable mu4e-alert notifications on laptop only."
      (when (eq my-device 'laptop)
        (require 'mu4e-alert)
        (mu4e-alert-enable-notifications)))
    (add-hook 'after-init-hook #'my/mu4e-alert-enable))

;;; Outgoing mail via msmtp
  (with-eval-after-load 'message
    (setq message-send-mail-function        'message-send-mail-with-sendmail
          sendmail-program                  "/usr/bin/msmtp"
          message-sendmail-extra-arguments  '("--read-envelope-from")
          message-sendmail-f-is-evil        t)))
