This file contains consolidated configurations for general Org-mode extensions, including attachments, smart TAB, hooks, capture templates, and LaTeX/PDF export processes. VLSI-specific diagram settings (TikZ, CircuitTikZ, ditaa) are in ~org/customxtn.org~ (optional). Tangled to ~org/orgxtn.el~ and loaded via ~my/tangle-if-needed~ in ~config.org~.

* Features:
- Centralized attachments directory
- Smart TAB integration (Yasnippet → CDLaTeX → Org-cycle)
- Vertical PDF preview
- Auto-update #+LAST_MODIFIED on save
- External archiving to dedicated `archive/` folder (keeps main Org/Roam files clean and fast)
- Simple Org-roam capture template loader
- Minimal Babel support (shell + LaTeX)
- LaTeX/PDF export via latexmk

  (defun my/org-smart-tab ()
    "Smart TAB: Yankpad → Yasnippet → CDLaTeX → Org-cycle."
    (interactive)
    (cond
     ((org-at-heading-p) (org-cycle))
     ((org-in-src-block-p) (org-cycle))
     ((org-at-table-p 'any) (org-table-next-field))
     ;; Yankpad trigger expansion (must come before yasnippet)
     ((and (fboundp 'yankpad-expand) (yankpad-expand)) t)
     ;; Yasnippet expansion
     ((or (and (fboundp 'yas-maybe-expand) (yas-maybe-expand))
          (and (fboundp 'yas-expand) (yas-expand))) t)
     ;; CDLaTeX in math mode
     ((and (or (bound-and-true-p cdlatex-mode) (bound-and-true-p org-cdlatex-mode))
           (org-inside-LaTeX-fragment-p)) (cdlatex-tab))
     ;; Default: cycle
     (t (org-cycle))))
    (define-key org-mode-map (kbd "TAB") #'my/org-smart-tab))

;; Step 1 (once per new target): M-x my-org-update-tag-targets
;; Step 2 (weekly, from inbox):  M-x my-org-refile-to-tag-targets

(defun my-org-update-tag-targets ()
  "Scan agenda files for :TAG_TARGET: and rebuild `my-org-refile-to-ids'.
Run once after adding or changing any TAG_TARGET property."
  (interactive)
  (setq my-org-refile-to-ids
        (let (list)
          (org-map-entries
           (lambda ()
             (cons (concat "+" (org-entry-get (point) "TAG_TARGET"))
                   (org-id-get-create)))
           "TAG_TARGET={.}"
           'agenda)))
  (customize-save-variable 'my-org-refile-to-ids my-org-refile-to-ids)
  (message "Tag targets rebuilt: %d rules" (length my-org-refile-to-ids)))

;; Performs the actual refiling by identifying matching headings and moving them to the target
  (defun my-org-refile-matches-to-heading (match target-heading-id &optional scope copy)
  "Refile matching headings to target."
  (if-let (target-marker (org-id-find target-heading-id t))
      (let* ((target-rfloc (with-current-buffer (marker-buffer target-marker)
                             (goto-char target-marker)
                             (list (org-get-heading)
                                   (buffer-file-name (marker-buffer target-marker))
                                   nil
                                   target-marker)))
             (headings-to-copy (org-map-entries (lambda () (point-marker)) match scope)))
        (mapc
         (lambda (heading-marker)
           (with-current-buffer (marker-buffer heading-marker)
             (goto-char heading-marker)
             (org-refile nil nil target-rfloc (when copy "Copy"))))
         (nreverse headings-to-copy))
        (message "%s %d headings!" (if copy "Copied" "Refiled") (length headings-to-copy)))
     (warn "Could not find target %S" target-heading-id)))

#+BEGIN_SRC emacs-lisp
;;; #+LAST_MODIFIED: auto-update on save
;;; Added: 2025-07-28 | Updated: 2026-05-12
;;; Fixes: eobp guard, presence-check, buffer-local hook, re-entrancy guard

(defvar my/org-update-last-modified--active nil
  "Non-nil while `my/org-update-last-modified' is running. Prevents re-entrancy.")

(defun my/org-update-last-modified ()
  "Update #+LAST_MODIFIED: on save, preserving the keyword case found in file."
  (when (and (derived-mode-p 'org-mode)
             (not (bound-and-true-p my/org-update-last-modified--active)))
    (let ((my/org-update-last-modified--active t))
      (save-excursion
        (let* ((case-fold-search t)
               (keyword
                (progn
                  (goto-char (point-min))
                  (if (re-search-forward "^\\(#\\+last_modified:\\)" nil t)
                      (match-string-no-properties 1)
                    "#+LAST_MODIFIED:")))
               (timestamp (format-time-string "%Y-%m-%d %a %H:%M")))
          ;; Only run if #+LAST_MODIFIED: is already declared in this file
          (goto-char (point-min))
          (when (re-search-forward "^#\\+last_modified:" nil t)
            (goto-char (point-min))
            (while (re-search-forward "^#\\+last_modified:.*\n?" nil t)
              (delete-region (match-beginning 0) (match-end 0)))
            (goto-char (point-min))
            (if (re-search-forward "^#\\+created:.*$" nil t)
                (progn
                  (end-of-line)
                  (insert (format "\n%s [%s]" keyword timestamp)))
              (goto-char (point-min))
              (let ((metadata-end (point-min)))
                ;; (not (eobp)) must be first: prevents infinite loop on
                ;; empty files where (looking-at "^[ \t]*$") matches forever
                (while (and (not (eobp))
                            (looking-at "^\\(#\\|[ \t]*$\\)"))
                  (setq metadata-end (line-end-position))
                  (forward-line 1))
                (goto-char metadata-end)
                (unless (bolp) (insert "\n"))
                (insert (format "%s [%s]\n" keyword timestamp))))))))))

;; nil t = do not prepend, t = buffer-local (only fires in org buffers)
(add-hook 'org-mode-hook
          (lambda ()
            (add-hook 'before-save-hook #'my/org-update-last-modified nil t)))
#+END_SRC

(defun my/org-glossary-safe-enable ()
  "Enable `org-glossary-mode' unless buffer is a glossary source file."

             (boundp 'org-glossary-global-terms)
             (not (member (expand-file-name buffer-file-name)
                          (ensure-list org-glossary-global-terms))))
    (org-glossary-mode 1)))

  (advice-add 'org-glossary--register-buffer-dependencies :around
  (lambda (orig &optional path-spec already-seen)

** Org glossaries to Latex glossaries conversion 
#+BEGIN_SRC emacs-lisp
;; ========================================
;; Org-glossary -> LaTeX .tex sync
;; Single source of truth: edit .org files only.
;; The .tex files are auto-derived on save.
;; Source : glossary/{gen,tech}_acronyms.org
;; Output : latex/{gen,tech}-acronyms.tex
;; Added  : 2026-04-20
;; ========================================

(defvar my/org-acronym-short-overrides
  ;; Extend only when (upcase key) gives the wrong SHORT form.
  '(("bsc"     . "BSc")
    ("msc"     . "MSc")
    ("phd"     . "PhD")
    ("poc"     . "PoC")
    ("pnr"     . "PnR")
    ("finfet"  . "FinFET")
    ("gaa-fet" . "GAAFET")
    ("fd-soi"  . "FD-SOI"))
  "Alist of (key . SHORT) for acronyms where `(upcase key)` is incorrect.")

(defun my/org-acronym-short (key)
  "Return SHORT display form for KEY.
Check `my/org-acronym-short-overrides' first, else fall back to `(upcase key)`."
  (or (cdr (assoc key my/org-acronym-short-overrides))
      (upcase key)))

(defun my/org-acronyms-to-tex (org-file tex-file)
  "Generate \\newacronym and \\newglossaryentry entries in TEX-FILE
from * Acronyms and * Glossary sections in ORG-FILE.
Acronyms -> \\acronymtype; Glossary -> main type (default).
Aborts with an error if any key appears in both sections.
Skips regeneration when TEX-FILE is up to date."
  (when (and (file-exists-p org-file)
             (or (not (file-exists-p tex-file))
                 (file-newer-than-file-p org-file tex-file)))
    (let (acronyms glossary)
      (with-temp-buffer
        (insert-file-contents org-file)
        (goto-char (point-min))

        ;; Each section parsed as "- key :: Full description" lines.
        (when (re-search-forward "^\\* Acronyms" nil t)
          (forward-line 1)
          (while (looking-at "^[ \t]*$") (forward-line 1))
          (while (looking-at "^- \\([^ ]+\\) :: \\(.*\\)")
            (push (list (match-string 1) (match-string 2)) acronyms)
            (forward-line 1)))

        (goto-char (point-min))
        (when (re-search-forward "^\\* Glossary" nil t)
          (forward-line 1)
          (while (looking-at "^[ \t]*$") (forward-line 1))
          (while (looking-at "^- \\([^ ]+\\) :: \\(.*\\)")
            (push (list (match-string 1) (match-string 2)) glossary)
            (forward-line 1))))

      ;; Guard: abort if any key exists in both sections.
      ;; Duplicate labels cause a fatal LaTeX error; fix the .org source.
      (let ((dupes (seq-filter (lambda (k) (assoc k glossary))
                               (mapcar #'car acronyms))))
        (when dupes
          (display-warning
           'org-glossary-sync
           (format "%s: duplicate key(s) in both Acronyms and Glossary: %s\nFix the source -- .tex NOT regenerated."
                   (file-name-nondirectory org-file) dupes)
           :error)
          (throw 'org-glossary-sync-abort nil)))

      (with-temp-file tex-file
        (insert "%% Auto-generated from "
                (file-name-nondirectory org-file) "\n")
        (insert "%% Do not edit manually -- edit the .org source instead\n\n")

        (when acronyms
          (insert "%% --- Acronyms (type: acronymtype) ---\n")
          (dolist (e (nreverse acronyms))
            (let* ((key   (nth 0 e))
                   (full  (nth 1 e))
                   (short (my/org-acronym-short key)))
              (insert (format "\\newacronym{%s}{%s}{%s}\n"
                              key short full))))
          (insert "\n"))

        ;; Glossary entries go into the main (default) glossary type,
        ;; separate from acronyms. Print on demand:
        ;;   #+latex: \printglossary[title={Glossary},style=index]
        (when glossary
          (insert "%% --- Glossary (type: main) ---\n")
          (dolist (e (nreverse glossary))
            (let* ((key  (nth 0 e))
                   (desc (nth 1 e)))
              (insert
               (format "\\newglossaryentry{%s}{name={%s},description={%s}}\n"
                       key key desc))))))

      (message "org-glossary sync: regenerated %s"
               (file-name-nondirectory tex-file)))))

(defun my/org-glossary-sync-tex ()
  "Regenerate .tex glossary file when a glossary source Org file is saved.
Resolves paths from `my/notes-root-dir'; no absolute paths."
  (when buffer-file-name
    (let* ((f      (expand-file-name buffer-file-name))
           (gdir   (expand-file-name "glossary/" my/notes-root-dir))
           (texdir (expand-file-name "latex/"    my/notes-root-dir)))
      ;; catch wraps the cond so a duplicate-key abort exits cleanly.
      (catch 'org-glossary-sync-abort
        (cond
         ((string= f (expand-file-name "gen_acronyms.org"  gdir))
          (my/org-acronyms-to-tex f (expand-file-name "gen-acronyms.tex"  texdir)))
         ((string= f (expand-file-name "tech_acronyms.org" gdir))
          (my/org-acronyms-to-tex f (expand-file-name "tech-acronyms.tex" texdir))))))))

;; Fires only on C-x C-s; silent for all non-glossary files.
(add-hook 'after-save-hook #'my/org-glossary-sync-tex)
